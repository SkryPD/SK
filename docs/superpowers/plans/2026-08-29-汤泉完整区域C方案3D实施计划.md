> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 依据《温泉、娱乐室平面.pdf》建立共享几何母版，交付一张 C 方案拆顶鸟瞰轴测图和一张同模型室内透视图。

**Architecture:** 用一个以毫米为单位的 Python 几何清单锁定男汤、女汤、过渡空间、绿化隔离带、设备房和两座 `8180 × 4240 mm` 池体；用 Pillow 从同一清单生成轴测控制图和室内透视控制图；再通过官方 imagegen CLI、`gpt-image-2` 和 `config.toml` 相同节点进行写实化。生成图必须通过原平面、控制图和两张成图的逐项人工核对，不能把 AI 输出当作施工级模型。

**Tech Stack:** Python 3.14、Pillow、stdlib `unittest`、OpenAI 官方 imagegen CLI、`gpt-image-2`、Git for Windows。

```python
from dataclasses import dataclass


@dataclass(frozen=True)
class Rect:
    x: int
    y: int
    width: int
    height: int
    z: int = 0

    @property
    def size(self):
        return self.width, self.height

    @property
    def right(self):
        return self.x + self.width

    @property
    def bottom(self):
        return self.y + self.height

    def contains(self, other):
        return (
            self.x <= other.x
            and self.y <= other.y
            and self.right >= other.right
            and self.bottom >= other.bottom
        )


CEILING_HEIGHT = 3200  # 设计暂定值，不是施工图尺寸
WALL_HEIGHT_CUTAWAY = 1700
WALL_THICKNESS = 140


def build_model():
    # 本地坐标：x 向东，y 向南；单位 mm。北侧设备房以负 y 表示。
    return {
        "equipment": Rect(0, -7260, 10000, 7260),
        "male_support": Rect(0, 0, 5038, 6580),
        "male_bath": Rect(5038, 0, 8680, 6580),
        "male_planter": Rect(13718, 0, 1000, 6580),
        "green_divider": Rect(5038, 6580, 9680, 140),
        "female_support": Rect(0, 6720, 5038, 6580),
        "female_bath": Rect(5038, 6720, 8680, 6580),
        "female_planter": Rect(13718, 6720, 1000, 6580),
        # 池体 8180 x 4240 来自平面图；池体在各自 8680 宽汤区内居中。
        "male_pool": Rect(5288, 1170, 8180, 4240, -600),
        "female_pool": Rect(5288, 7890, 8180, 4240, -600),
        # 门洞只锁定平面位置，门高留给设计暂定值。
        "male_entry": Rect(0, 2550, WALL_THICKNESS, 1480),
        "female_entry": Rect(0, 9270, WALL_THICKNESS, 1480),
        "male_bath_door": Rect(4898, 2550, 280, 1480),
        "female_bath_door": Rect(4898, 9270, 280, 1480),
    }


def validate_model(model):
    errors = []
    for pool_name, bath_name in (
        ("male_pool", "male_bath"),
        ("female_pool", "female_bath"),
    ):
        if model[pool_name].size != (8180, 4240):
            errors.append(f"{pool_name} dimension drift")
        if not model[bath_name].contains(model[pool_name]):
            errors.append(f"{pool_name} outside {bath_name}")
    if model["green_divider"].height != 140:
        errors.append("green divider dimension drift")
    if model["equipment"].bottom != 0:
        errors.append("equipment room must adjoin north edge")
    if model["female_support"].y != model["green_divider"].bottom:
        errors.append("female zone must adjoin divider")
    return errors
```

```python
import math
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

from .geometry import CEILING_HEIGHT


CANVAS = (2048, 1152)
BG = (244, 241, 234)
INK = (48, 48, 45)
WALL = (221, 216, 205)
STONE = (208, 202, 191)
WATER = (125, 182, 184)
GREEN = (100, 137, 92)
WOOD = (174, 132, 91)


def _font(size=30):
    candidates = [
        r"C:\Windows\Fonts\msyh.ttc",
        r"C:\Windows\Fonts\simhei.ttf",
    ]
    for candidate in candidates:
        if Path(candidate).exists():
            return ImageFont.truetype(candidate, size)
    return ImageFont.load_default()


def _iso(x_mm, y_mm, z_mm, origin=(980, 210), scale=0.043):
    x, y, z = x_mm * scale, y_mm * scale, z_mm * scale
    return (
        origin[0] + (x - y) * math.cos(math.radians(30)),
        origin[1] + (x + y) * math.sin(math.radians(30)) - z,
    )


def _rect_poly(rect, z=0):
    return [
        _iso(rect.x, rect.y, z),
        _iso(rect.right, rect.y, z),
        _iso(rect.right, rect.bottom, z),
        _iso(rect.x, rect.bottom, z),
    ]


def render_axonometric(model, output_path):
    image = Image.new("RGB", CANVAS, BG)
    draw = ImageDraw.Draw(image)
    floor_names = [
        "equipment", "male_support", "male_bath", "male_planter",
        "green_divider", "female_support", "female_bath", "female_planter",
    ]
    for name in floor_names:
        rect = model[name]
        color = GREEN if "green" in name or "planter" in name else STONE
        draw.polygon(_rect_poly(rect), fill=color, outline=INK, width=3)
    for name in ("male_pool", "female_pool"):
        rect = model[name]
        draw.polygon(_rect_poly(rect, 30), fill=WATER, outline=INK, width=4)
    # 低墙只用于读取空间，不表示真实拆改。
    for name in ("male_support", "male_bath", "female_support", "female_bath", "equipment"):
        rect = model[name]
        base = _rect_poly(rect)
        top = _rect_poly(rect, 1700)
        for i in range(4):
            j = (i + 1) % 4
            draw.line([base[i], top[i]], fill=INK, width=3)
            draw.line([top[i], top[j]], fill=INK, width=3)
    font = _font(34)
    labels = {
        "male_bath": "男汤",
        "female_bath": "女汤",
        "male_support": "男汤过渡/洗浴",
        "female_support": "女汤过渡/洗浴",
        "equipment": "设备房外壳",
    }
    for name, label in labels.items():
        rect = model[name]
        px = _iso(rect.x + rect.width / 2, rect.y + rect.height / 2, 90)
        draw.text(px, label, font=font, fill=INK, anchor="mm")
    output_path = Path(output_path)
    output_path.parent.mkdir(parents=True, exist_ok=True)
    image.save(output_path)


def _camera_project(point, camera, target, focal=1150):
    px, py, pz = (v / 1000 for v in point)
    cx, cy, cz = (v / 1000 for v in camera)
    tx, ty, tz = (v / 1000 for v in target)
    forward = (tx - cx, ty - cy, tz - cz)
    flen = math.sqrt(sum(v * v for v in forward))
    forward = tuple(v / flen for v in forward)
    right = (forward[1], -forward[0], 0.0)
    rlen = math.sqrt(sum(v * v for v in right))
    right = tuple(v / rlen for v in right)
    up = (
        right[1] * forward[2],
        -right[0] * forward[2],
        right[0] * forward[1] - right[1] * forward[0],
    )
    delta = (px - cx, py - cy, pz - cz)
    depth = max(0.2, sum(delta[i] * forward[i] for i in range(3)))
    sx = sum(delta[i] * right[i] for i in range(3))
    sy = sum(delta[i] * up[i] for i in range(3))
    return 1024 + focal * sx / depth, 610 - focal * sy / depth


def render_interior(model, output_path):
    image = Image.new("RGB", CANVAS, (234, 229, 219))
    draw = ImageDraw.Draw(image)
    camera = (3600, 3290, 1550)
    target = (11000, 3000, 1100)
    project = lambda p: _camera_project(p, camera, target)
    bath = model["male_bath"]
    pool = model["male_pool"]
    # 室内壳体：地面、两侧墙、顶面、尽端墙。
    floor = [(bath.x, bath.y, 0), (bath.right, bath.y, 0), (bath.right, bath.bottom, 0), (bath.x, bath.bottom, 0)]
    ceiling = [(bath.x, bath.y, CEILING_HEIGHT), (bath.right, bath.y, CEILING_HEIGHT), (bath.right, bath.bottom, CEILING_HEIGHT), (bath.x, bath.bottom, CEILING_HEIGHT)]
    draw.polygon([project(p) for p in floor], fill=STONE, outline=INK)
    draw.polygon([project(p) for p in ceiling], fill=WOOD, outline=INK)
    for y in (bath.y, bath.bottom):
        wall = [(bath.x, y, 0), (bath.right, y, 0), (bath.right, y, CEILING_HEIGHT), (bath.x, y, CEILING_HEIGHT)]
        draw.polygon([project(p) for p in wall], fill=WALL, outline=INK)
    far_wall = [(bath.right, bath.y, 0), (bath.right, bath.bottom, 0), (bath.right, bath.bottom, CEILING_HEIGHT), (bath.right, bath.y, CEILING_HEIGHT)]
    draw.polygon([project(p) for p in far_wall], fill=(214, 221, 205), outline=INK)
    water = [(pool.x, pool.y, 30), (pool.right, pool.y, 30), (pool.right, pool.bottom, 30), (pool.x, pool.bottom, 30)]
    draw.polygon([project(p) for p in water], fill=WATER, outline=INK, width=5)
    font = _font(28)
    draw.text((56, 55), "男汤室内透视控制图｜净高暂定 3.2m", font=font, fill=INK)
    draw.text((56, 95), "池体 8180 × 4240mm｜入口机位向东", font=_font(22), fill=INK)
    output_path = Path(output_path)
    output_path.parent.mkdir(parents=True, exist_ok=True)
    image.save(output_path)
```
