## 固定数据口径

工作簿初始输入采用以下可编辑暂估值，并在单元格备注中标明依据：

- 汤池数量 `2`；每池内平面 `8.18 × 4.24 m`。
- 池底至池沿完成面 `0.85 m`；坐台/台阶展开面暂估 `18.00 m²`，状态为“剖面暂估，需现场/深化复核”。
- 每个汤池区 `8.68 × 6.58 m`；每个休息/过渡区 `5.038 × 6.58 m`。
- 池沿计算宽度暂按 `0.25 m`；池外汤区地面为汤池区面积减池底投影和池沿面积。
- 房间贴砖外周长按两个 `13.718 × 6.58 m` 区域的周长之和暂估；墙高 `3.20 m`；两个入口洞口各按 `1.48 × 2.10 m` 暂扣。门窗、设备洞口及墙柱展开须现场复尺。
- 吊顶毛面积按完整区域 `14.718 × 13.30 m = 195.7494 m²` 暂估，工作簿显示约 `196 m²`。
- 端景肌理墙暂估 `16.00 m²`，从普通墙砖净面积中扣除，避免同一墙面重复计价。
- 木纹铝板暂按吊顶毛面积的 `15%` 计作“木纹转印升级差价”，不再重复计取一套完整吊顶板。
- 检修口暂估 `6 套`，线性灯驱动暂估 `6 套`；位置和功率分段须与机电、灯具及吊顶厂家深化。
- 基础筒灯暂估 `32 套`、隐藏线性灯暂估 `40 m`、端景洗墙灯暂估 `4 套`；这些数量只用于预算占位，必须经照度与电气设计复核。

// src/spa_procurement/quantity_model.mjs
export const defaultQuantityInputs = Object.freeze({
  poolCount: 2,
  poolLengthM: 8.18,
  poolWidthM: 4.24,
  poolFinishedDepthM: 0.85,
  copingWidthM: 0.25,
  bathZoneLengthM: 8.68,
  bathZoneWidthM: 6.58,
  supportZoneLengthM: 5.038,
  supportZoneWidthM: 6.58,
  roomCount: 2,
  wallHeightM: 3.2,
  entryCount: 2,
  entryWidthM: 1.48,
  entryHeightM: 2.1,
  poolStepAllowanceM2: 18,
  featureWallTileM2: 16,
  ceilingGrossWidthM: 14.718,
  ceilingGrossLengthM: 13.3,
  woodgrainCeilingRatio: 0.15,
  accessHatchesEach: 6,
  downlightsEach: 32,
  linearLightM: 40,
  wallWashersEach: 4,
  lightingDriversEach: 6,
});

const round4 = (value) => Math.round((value + Number.EPSILON) * 10000) / 10000;

export function calculateQuantities(inputs = defaultQuantityInputs) {
  const poolBottomM2 = inputs.poolCount * inputs.poolLengthM * inputs.poolWidthM;
  const poolWallM2 =
    inputs.poolCount *
    2 *
    (inputs.poolLengthM + inputs.poolWidthM) *
    inputs.poolFinishedDepthM;
  const poolCopingM2 =
    inputs.poolCount *
    ((inputs.poolLengthM + 2 * inputs.copingWidthM) *
      (inputs.poolWidthM + 2 * inputs.copingWidthM) -
      inputs.poolLengthM * inputs.poolWidthM);
  const bathZoneM2 =
    inputs.poolCount * inputs.bathZoneLengthM * inputs.bathZoneWidthM;
  const bathDeckFloorM2 = bathZoneM2 - poolBottomM2 - poolCopingM2;
  const supportFloorM2 =
    inputs.roomCount * inputs.supportZoneLengthM * inputs.supportZoneWidthM;
  const roomLengthM = inputs.supportZoneLengthM + inputs.bathZoneLengthM;
  const tiledPerimeterM =
    inputs.roomCount * 2 * (roomLengthM + inputs.bathZoneWidthM);
  const entryDeductionM2 =
    inputs.entryCount * inputs.entryWidthM * inputs.entryHeightM;
  const wallTileGrossM2 = tiledPerimeterM * inputs.wallHeightM - entryDeductionM2;
  const generalWallTileM2 = wallTileGrossM2 - inputs.featureWallTileM2;
  const ceilingGrossM2 = inputs.ceilingGrossWidthM * inputs.ceilingGrossLengthM;

  return Object.freeze({
    poolBottomM2: round4(poolBottomM2),
    poolWallM2: round4(poolWallM2),
    poolCopingM2: round4(poolCopingM2),
    bathDeckFloorM2: round4(bathDeckFloorM2),
    supportFloorM2: round4(supportFloorM2),
    wallTileGrossM2: round4(wallTileGrossM2),
    generalWallTileM2: round4(generalWallTileM2),
    featureWallTileM2: inputs.featureWallTileM2,
    ceilingGrossM2: round4(ceilingGrossM2),
    woodgrainUpgradeM2: round4(ceilingGrossM2 * inputs.woodgrainCeilingRatio),
    poolStepAllowanceM2: inputs.poolStepAllowanceM2,
    accessHatchesEach: inputs.accessHatchesEach,
    lighting: Object.freeze({
      downlightsEach: inputs.downlightsEach,
      linearLightM: inputs.linearLightM,
      wallWashersEach: inputs.wallWashersEach,
      lightingDriversEach: inputs.lightingDriversEach,
    }),
  });
}
```

// src/spa_procurement/market_sources.mjs
const allowedTiers = new Set(["mid", "economy"]);

export const requiredCategoryUnits = Object.freeze({
  pool_floor_tile: "m²",
  pool_wall_tile: "m²",
  pool_coping_tile: "m²",
  deck_floor_tile: "m²",
  wall_tile: "m²",
  feature_tile: "m²",
  ceiling_system: "m²",
  woodgrain_upgrade: "m²",
  linear_light_channel: "m",
  ceiling_access_hatch: "套",
  moisture_downlight: "套",
  ip67_linear_light: "m",
  wall_washer: "套",
  lighting_driver: "套",
  tile_installation: "m²",
  ceiling_installation: "m²",
  point_light_installation: "套",
  linear_light_installation: "m",
});

export function validateMarketSources(records, categoryUnits, asOfDate) {
  const errors = [];
  const cutoff = new Date(`${asOfDate}T00:00:00+08:00`);
  const categoryTiers = new Map(Object.keys(categoryUnits).map((name) => [name, new Set()]));
  const seenKeys = new Set();

  records.forEach((record, index) => {
    const prefix = `record ${index + 1}`;
    for (const field of [
      "category",
      "tier",
      "brand",
      "productOrEquivalent",
      "unit",
      "priceLow",
      "priceHigh",
      "sourceUrl",
      "checkedOn",
      "priceBasis",
      "regionNote",
    ]) {
      if (record[field] === undefined || record[field] === "") {
        errors.push(`${prefix}: missing ${field}`);
      }
    }
    if (!allowedTiers.has(record.tier)) errors.push(`${prefix}: invalid tier`);
    if (!(record.priceLow > 0 && record.priceHigh >= record.priceLow)) {
      errors.push(`${prefix}: invalid price range`);
    }
    if (!/^https:\/\//.test(record.sourceUrl ?? "")) {
      errors.push(`${prefix}: sourceUrl must use https`);
    }
    const checked = new Date(`${record.checkedOn}T00:00:00+08:00`);
    const ageDays = Math.floor((cutoff - checked) / 86400000);
    if (!Number.isFinite(ageDays) || ageDays < 0 || ageDays > 120) {
      errors.push(`${prefix}: checkedOn outside 120-day window`);
    }
    if (!(record.category in categoryUnits)) {
      errors.push(`${prefix}: unknown category`);
    } else if (record.unit !== categoryUnits[record.category]) {
      errors.push(`${prefix}: unit mismatch`);
    }
    const key = `${record.category}|${record.tier}`;
    if (seenKeys.has(key)) errors.push(`${prefix}: duplicate category-tier key`);
    seenKeys.add(key);
    categoryTiers.get(record.category)?.add(record.tier);
  });

  for (const [category, tiers] of categoryTiers) {
    for (const tier of allowedTiers) {
      if (!tiers.has(tier)) errors.push(`${category}: missing ${tier} option`);
    }
  }
  return { records, errors };
}
```
